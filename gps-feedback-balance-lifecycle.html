스코어 피드백 알고리즘

function applyFeedback(product, actionResult) { const effect = measureEffect(product, actionResult) // 1. Gap 변화량 측정 const gapDelta = effect.prevGap - effect.currentGap const improvementRate = gapDelta / effect.prevGap // 2. 액션 효과 계수 산출 (0.0 ~ 1.5) let effectCoeff = 1.0 if (improvementRate >= 0.5) effectCoeff = 1.3 // 대폭 개선 if (improvementRate >= 0.3) effectCoeff = 1.1 // 적정 개선 if (improvementRate < 0.1) effectCoeff = 0.8 // 미미작 → 우선도 하락 if (improvementRate < 0) effectCoeff = 0.6 // 악화 → 우선도 급락 // 3. 해당 축 Priority 가중치 보정 const axis = actionResult.targetAxis product.axisWeight[axis] *= effectCoeff normalizeWeights(product) // 총합 100% 유지 // 4. 상품 신뢰도 스코어 갑신 if (actionResult.slaBreached) { product.reliabilityScore -= 5 // SLA 위반 패널티 } if (actionResult.repeatedFail >= 3) { product.escalationFlag = true // 3회 연속 미개선 → 에스컬레이션 } return recalcGPS(product) }
피드백 규칙 상세
규칙 조건 GPS 반영 비고
성공 보상 Gap 개선률 ≥50% 해당축 가중치 ×1.3, 스코어 +5pt 보너스 유효 액션 패턴 강화
적정 개선 Gap 개선률 10~49% 해당축 가중치 ×1.1 기본 양성 피드백
미개선 Gap 개선률 <10% 해당축 우선도 ×0.8, 다른축으로 재배분 비효율 축은 자동 우선순위 하락
악화 Gap 증가 (개선률 <0) 우선도 ×0.6 + 알림 발송 근본원인 분석 필요
SLA 위반 미처리 시간 초과 신뢰도 -5pt, 해당상품 gap +10 에스컬레이션 프로세스 연동

난이도 계수 테스크별 소요 시간 계수 S:1.0 / A:0.6 / B:0.3 / C:0.1 S등급 1건 = A 1.67건 = B 3.3건 등가
경력 보정 MD 경력에 따른 처리 효율 신입 ×0.7 / 중급 ×1.0 / 시니어 ×1.3 시니어 MD는 30% 더 처리 가능
카테고리 전문성 담당 카테고리 상품은 효율 보너스 전문 ×1.2 / 비전문 ×0.8 타 카테고리 상품 배정 시 패널티
일과 부하 현재 진행중 테스크 양 0~100% (실시간 갑신) 80% 이상 시 신규 배정 차단
function calcMDCapacity(md) { const BASE_SLOTS = 15 const expMult = { junior: 0.7, mid: 1.0, senior: 1.3 } const specMult = md.isSpecialist ? 1.2 : 0.8 const rawCapacity = BASE_SLOTS * expMult[md.level] * specMult // 현재 부하 차감 const currentLoad = md.activeTasks.reduce((sum, t) => sum + DIFFICULTY[t.tier], 0) const available = rawCapacity - currentLoad return { totalCapacity: rawCapacity, currentLoad, available: Math.max(0, available), loadPercent: (currentLoad / rawCapacity * 100).toFixed(1), isOverloaded: currentLoad > rawCapacity * 0.8 } }
배분 알고리즘
상품 배정 시 MD 워크로드를 균등하게 분산하는 로직
배분 우선순위
1. 카테고리 전문 MD 우선
해당 상품의 카테고리를 담당하는 MD에게 우선 배정. 전문성 보너스 ×1.2 적용
2. 여유 역량 체크
MD loadPercent < 80% 인지 확인. 초과 시 다음 MD로 넘김
3. 난이도 균형 체크
S등급 테스크가 특정 MD에게 집중되지 않도록 분산. MD당 S등급 최대 5건
4. 교차 배정
타 카테고리 MD에게 배정 시 비전문 패널티(×0.8) 적용. 기여손실 내림차순으로 우선
5. 긴급 오버플로우 처리
모든 MD가 80% 이상일 때: 가장 낮은 부하 MD에게 강제 배정 + 팀리드 알림
6. 전체 과부하 에스컬레이션
팀 평균 부하 ≥90%: B/C등급 SLA 연장 + 본부장 리포트 자동생성

OVERLOAD MD 부하 ≥100% B/C 테스크 중 기여손실 최소 상품 자동 재배분 실시간 감지

function detectLifecycle(product) { const daysSinceLaunch = daysBetween(product.launchDate, today) const mom = product.gmvMoM // 월간 GMV 성장률 const trend = product.gmvTrend3M // 3개월 추세 // 신규 도입기: 출시 30일 미만 if (daysSinceLaunch < 30) return 'NEW' // 성장기: MoM ≥ 10% && 3개월 우상향 추세 if (mom >= 10 && trend === 'UP') return 'GROWTH' // 성숙기: MoM -5% ~ +10% && 안정 추세 if (mom >= -5 && mom < 10 && trend === 'STABLE') return 'MATURE' // 쇄퇴기: MoM < -5% 또는 3개월 하락 추세 if (mom < -5 || trend === 'DOWN') { // 터출 판단: 3개월 연속 하락 + MoM < -15% if (product.consecutiveDecline >= 3 && mom < -15) { return 'EXIT' } return 'DECLINE' } return 'MATURE' // 기본값 }

function seasonalAdjust(product, rawLifecycle) { const seasonal = getSeasonalIndex(product.category, currentMonth) // seasonal: 0.7(비시즌) ~ 1.0(보통) ~ 1.5(피크) // 비시즌 하락을 쇄퇴로 오판하지 않음 if (seasonal < 0.85 && rawLifecycle === 'DECLINE') { const yoyGrowth = product.gmvYoY // 전년 동기 대비 if (yoyGrowth >= -5) { return 'MATURE' // 시즐널 비시즌일 벀 → 성숙기로 유지 } } // 피크 시즐 성장을 성장기로 오판하지 않음 if (seasonal > 1.2 && rawLifecycle === 'GROWTH') { const yoyGrowth = product.gmvYoY if (yoyGrowth < 5) { return 'MATURE' // 시즐널 붙 때문 → 성숙기로 유지 } } return rawLifecycle }
시즐널 인덱스 기준: 카테고리별 월별 시즐널 계수는 전년 동월 GMV 대비 비율로 산출. 예: 패션 3월 계수=1.3 (봄시즌 피크), 마트 1월 계수=1.4 (설날 수요). 최소 2년 데이터 필요.
